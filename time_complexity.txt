// 1) Sort and remove duplicates elements 
// best sloution code using sort() & unique() -> O(N log N)
->
#include <iostream>
#include <algorithm>
using namespace std;

int main()
{
    int arr[] = {5, 5,8,3,11,678,789,23,34,34,34,56,67,11,23,2, 8, 2, 1, 5, 8, 3};
    int n = sizeof(arr) / sizeof(arr[0]);

    sort(arr, arr + n);
    int newSize = unique(arr, arr + n) - arr;
    
    cout << "After removing duplicates:\n";
    for(int i = 0; i < newSize; i++)
    {
        cout << arr[i] << " ";
    }

    return 0;
}
