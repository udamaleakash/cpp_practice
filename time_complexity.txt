Complexity Ranking (Best → Worst)

O(1)
   ↓
O(log N) -> Binary search
   ↓
O(N) -> Linear search
   ↓
O(N log N)
   ↓
O(N²)
   ↓
O(2^N)
   ↓
O(N!)


Ranking (Fastest to Slowest Search)
unordered_map  -> O(1)

map            -> O(log N)

array          -> O(N)

vector         -> O(N)

2D array       -> O(N²)  (if full scan required)


// sort and remove duplicate elements
// Vector + sort() + unique() + erase()
// Time complixity - O(N log N)

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

int main()
{
    int arr[] = {5,5,8,3,11,678,789,23,34,34,34,56,67,11,23,2,8,2,1,5,8,3};
    int n = sizeof(arr) / sizeof(arr[0]);

    vector<int> v(arr, arr + n);

    sort(v.begin(), v.end());

    v.erase(unique(v.begin(), v.end()), v.end());

    cout << "Unique Elements:\n";

    for (int x : v)
        cout << x << " ";

    return 0;
}
